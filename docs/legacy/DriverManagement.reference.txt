import React, { useEffect, useState } from "react";
import { supabase } from "../services/supabase";
import { Bike, UserPlus, Phone, ShieldCheck } from "lucide-react";

export function DriverManagement() {
  const [drivers, setDrivers] = useState<any[]>([]);
  const [loading, setLoading] = useState(false);

  // Form states
  const [fullName, setFullName] = useState("");
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [phone, setPhone] = useState("");

  const fetchDrivers = async () => {
    const { data } = await (supabase as any)
      .from("profiles")
      .select("*")
      .eq("role", "driver")
      .order("created_at", { ascending: false });
    if (data) setDrivers(data);
  };

  useEffect(() => {
    fetchDrivers();
  }, []);

  const handleAddDriver = async (e: React.FormEvent) => {
    e.preventDefault();
    setLoading(true);

    // 1. Dùng quyền Admin (Service Role) để tạo tài khoản Auth ẩn danh
    const { data: authData, error: authError } = await (
      supabase as any
    ).auth.admin.createUser({
      email: email,
      password: password,
      email_confirm: true,
    });

    if (authError) {
      alert("Lỗi tạo tài khoản Auth: " + authError.message);
      setLoading(false);
      return;
    }

    // 2. Chèn bản ghi vào bảng profiles với role = 'driver'
    // 2. Cập nhật bản ghi vào bảng profiles với role = 'driver'
    if (authData.user) {
      const { error: profileError } = await (supabase as any)
        .from("profiles")
        .upsert({
          id: authData.user.id,
          full_name: fullName,
          phone: phone,
          role: "driver",
        });

      if (profileError) {
        alert("Lỗi tạo hồ sơ tài xế: " + profileError.message);
      } else {
        alert("Thêm tài xế thành công!");
        setFullName("");
        setEmail("");
        setPassword("");
        setPhone("");
        fetchDrivers();
      }
    }
    setLoading(false);
  };

  return (
    <div className="grid grid-cols-1 md:grid-cols-3 gap-6 p-6">
      {/* CỘT TRÁI: FORM THÊM TÀI XẾ */}
      <div className="md:col-span-1 bg-white p-6 rounded-2xl shadow-sm border border-gray-100 h-fit">
        <div className="flex items-center gap-2 mb-6">
          <div className="h-8 w-8 rounded-full bg-teal-50 text-teal-600 flex items-center justify-center">
            <UserPlus className="w-4 h-4" />
          </div>
          <h3 className="font-bold text-slate-800">Cấp Tài Khoản Tài Xế</h3>
        </div>

        <form onSubmit={handleAddDriver} className="space-y-4">
          <div>
            <label className="block text-xs font-semibold text-slate-600 mb-1">
              Họ và tên *
            </label>
            <input
              required
              className="w-full border border-slate-200 p-2.5 rounded-xl bg-slate-50 text-sm focus:ring-2 focus:ring-teal-500 outline-none"
              placeholder="Nguyễn Văn A"
              value={fullName}
              onChange={(e) => setFullName(e.target.value)}
            />
          </div>
          <div>
            <label className="block text-xs font-semibold text-slate-600 mb-1">
              Số điện thoại *
            </label>
            <input
              required
              className="w-full border border-slate-200 p-2.5 rounded-xl bg-slate-50 text-sm focus:ring-2 focus:ring-teal-500 outline-none"
              placeholder="0912345678"
              value={phone}
              onChange={(e) => setPhone(e.target.value)}
            />
          </div>
          <div>
            <label className="block text-xs font-semibold text-slate-600 mb-1">
              Email đăng nhập *
            </label>
            <input
              required
              type="email"
              className="w-full border border-slate-200 p-2.5 rounded-xl bg-slate-50 text-sm focus:ring-2 focus:ring-teal-500 outline-none"
              placeholder="taixe1@gmail.com"
              value={email}
              onChange={(e) => setEmail(e.target.value)}
            />
          </div>
          <div>
            <label className="block text-xs font-semibold text-slate-600 mb-1">
              Mật khẩu *
            </label>
            <input
              required
              type="password"
              className="w-full border border-slate-200 p-2.5 rounded-xl bg-slate-50 text-sm focus:ring-2 focus:ring-teal-500 outline-none"
              placeholder="Ít nhất 6 ký tự"
              value={password}
              onChange={(e) => setPassword(e.target.value)}
            />
          </div>

          <button
            type="submit"
            disabled={loading}
            className="w-full mt-2 bg-teal-600 hover:bg-teal-700 disabled:bg-slate-300 text-white font-bold py-3 px-4 rounded-xl text-sm transition-all"
          >
            {loading ? "Đang tạo..." : "+ Xác Nhận Tạo Tài Xế"}
          </button>
        </form>
      </div>

      {/* CỘT PHẢI: DANH SÁCH TÀI XẾ */}
      <div className="md:col-span-2 bg-white p-6 rounded-2xl shadow-sm border border-gray-100">
        <div className="flex items-center gap-2 mb-6">
          <div className="h-8 w-8 rounded-full bg-indigo-50 text-indigo-600 flex items-center justify-center">
            <Bike className="w-4 h-4" />
          </div>
          <h3 className="font-bold text-slate-800">
            Đội Ngũ Tài Xế Đang Hoạt Động ({drivers.length})
          </h3>
        </div>

        {drivers.length === 0 ? (
          <div className="text-center py-12 border-2 border-dashed border-slate-200 rounded-xl">
            <p className="text-sm text-slate-400">
              Chưa có tài xế nào trong hệ thống.
            </p>
          </div>
        ) : (
          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
            {drivers.map((driver) => (
              <div
                key={driver.id}
                className="p-4 border border-slate-100 bg-slate-50 rounded-xl flex gap-3 items-center"
              >
                <img
                  src={
                    driver.avatar_url ||
                    "https://images.unsplash.com/photo-1500648767791-00dcc994a43e?auto=format&fit=crop&w=150&q=80"
                  }
                  alt="avatar"
                  className="w-12 h-12 rounded-full object-cover border border-slate-200"
                />
                <div className="flex-1">
                  <h4 className="font-bold text-slate-800 text-sm">
                    {driver.full_name}
                  </h4>
                  <div className="flex items-center gap-1 text-slate-500 text-xs mt-0.5">
                    <Phone className="w-3 h-3" />{" "}
                    {driver.phone || "Chưa cập nhật SĐT"}
                  </div>
                  <div className="flex items-center gap-1 text-emerald-600 font-semibold text-xs mt-1">
                    <ShieldCheck className="w-3 h-3" /> Đối tác tài xế
                  </div>
                </div>
              </div>
            ))}
          </div>
        )}
      </div>
    </div>
  );
}
